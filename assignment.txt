8. Prime Number

import 'dart:io';

void main() {
  int n = int.parse(stdin.readLineSync()!);
  bool prime = true;

  if (n < 2) {
    prime = false;
  } else {
    for (int i = 2; i <= n ~/ 2; i++) {
      if (n % i == 0) {
        prime = false;
        break;
      }
    }
  }

  print(prime ? "Prime" : "Not Prime");
}

10. Reverse a Number

import 'dart:io';

void main() {
  int n = int.parse(stdin.readLineSync()!);
  int reverse = 0;

  while (n != 0) {
    int digit = n % 10;
    reverse = reverse * 10 + digit;
    n ~/= 10;
  }

  print(reverse);
}

15. Armstrong Number

import 'dart:io';

void main() {
  int n = int.parse(stdin.readLineSync()!);
  int original = n;
  int temp = n;
  int digits = n.toString().length;
  int sum = 0;

  while (temp != 0) {
    int digit = temp % 10;
    int power = 1;

    for (int i = 0; i < digits; i++) {
      power *= digit;
    }

    sum += power;
    temp ~/= 10;
  }

  print(sum == original ? "Armstrong Number" : "Not Armstrong Number");
}

18. Count Vowels and Consonants

import 'dart:io';

void main() {
  String str = stdin.readLineSync()!.toLowerCase();
  int vowels = 0;
  int consonants = 0;

  for (int i = 0; i < str.length; i++) {
    String ch = str[i];

    if ("aeiou".contains(ch)) {
      vowels++;
    } else if (ch.codeUnitAt(0) >= 97 && ch.codeUnitAt(0) <= 122) {
      consonants++;
    }
  }

  print("Vowels: $vowels");
  print("Consonants: $consonants");
}

26. Sort Ascending

      .split(" ")

      .toList();

  for (int i = 0; i < list.length - 1; i++) {
    for (int j = 0; j < list.length - i - 1; j++) {
      if (list[j] > list[j + 1]) {

29. Second Largest Element

import 'dart:io';

void main() {
  List<int> list = stdin.readLineSync()!
      .split(" ")
      .map(int.parse)
      .toList();

  list.sort();

  int largest = list[list.length - 1];
  int secondLargest = -999999999;

  for (int i = list.length - 2; i >= 0; i--) {
    if (list[i] != largest) {
      secondLargest = list[i];
      break;
    }
  }

  if (secondLargest == -999999999) {
    print("No second largest element");
  } else {
    print(secondLargest);
  }
}

32. Function to Check Prime

import 'dart:io';

bool isPrime(int n) {
  if (n < 2) {
    return false;
  }

  for (int i = 2; i <= n ~/ 2; i++) {
    if (n % i == 0) {
      return false;
    }
  }

  return true;
}

void main() {
  int n = int.parse(stdin.readLineSync()!);
  print(isPrime(n) ? "Prime" : "Not Prime");
}

34. Recursive Factorial

import 'dart:io';

int factorial(int n) {
  if (n <= 1) {
    return 1;
  }

  return n * factorial(n - 1);
}

void main() {
  int n = int.parse(stdin.readLineSync()!);
  print(factorial(n));
}
